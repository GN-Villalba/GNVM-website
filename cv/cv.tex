% ================= RESEARCH IN PROGRESS =================
\section{Research in Progress}
\textbf{\small PhD Chapters}
\begin{itemize}
  \item \textbf{Raising the Floor: Minimum Quality Standards and the Training of Future Teachers in Colombia.}\\
  Evaluates the effects of mandatory minimum quality standards for teacher-training programmes on student outcomes and higher-education provision, using linked administrative data and a staggered difference-in-differences and event-study design.
  \item \textbf{What Students Want and Where They End Up: Aspirations, Local Options and Undermatch in Colombia.}\\
  Develops programme-level measures of quality and mismatch and a benchmark for students' stated aspirations, to compare the opportunities students aimed for with those they ultimately obtained, using linked administrative records.
\end{itemize}

\vspace{0.6ex}
\textbf{\small Working Papers}
\begin{itemize}
  \item \textbf{Gender Pay Gaps in a Middle-Income Country: Household Roles and the Motherhood Penalty.}\\
  Examines gender wage inequalities across the earnings distribution, with a focus on household roles and the motherhood penalty, using nationally representative household survey data. \textit{Developed from MSc dissertation, University of Sussex.}
  \item \textbf{Women-Only Admission Tracks in Japanese Universities: Effects, Selectivity and Limits} \textmd{(with Tomoto Masuda).}
  \item \textbf{Pro Rata, Not Parity: The Allocation of Women's Quotas in Japanese STEM} \textmd{(with Tomoto Masuda).}
\end{itemize}
